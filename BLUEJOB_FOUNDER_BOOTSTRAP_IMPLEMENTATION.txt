// server/src/bootstrapFounder.js
// One-time secure founder bootstrap for BlueJob.
// IMPORTANT: set FOUNDER_EMAIL and FOUNDER_TEMP_PASSWORD as Cloudflare runtime secrets.
// Do not commit the actual password into this file.

export async function bootstrapFounder({ db, hashPassword, env }) {
  const email = String(env.FOUNDER_EMAIL || "").trim().toLowerCase();
  const tempPassword = String(env.FOUNDER_TEMP_PASSWORD || "");

  if (!email || !tempPassword) {
    return { skipped: true, reason: "Founder bootstrap secrets not configured." };
  }

  if (tempPassword.length < 8) {
    throw new Error("FOUNDER_TEMP_PASSWORD must be at least 8 characters.");
  }

  // Use the SAME password hashing function already used by BlueJob login.
  const passwordHash = await hashPassword(tempPassword);

  await db.query("BEGIN");

  try {
    const existing = await db.query(
      `SELECT id, email, role
         FROM users
        WHERE LOWER(email) = $1
        LIMIT 1`,
      [email]
    );

    let user;

    if (existing.rows.length) {
      const updated = await db.query(
        `UPDATE users
            SET password_hash = $1,
                role = 'SUPER_ADMIN',
                must_change_password = TRUE,
                updated_at = NOW()
          WHERE id = $2
          RETURNING id, email, role, must_change_password`,
        [passwordHash, existing.rows[0].id]
      );

      user = updated.rows[0];
    } else {
      const inserted = await db.query(
        `INSERT INTO users (
            email,
            password_hash,
            role,
            must_change_password,
            created_at,
            updated_at
         )
         VALUES ($1, $2, 'SUPER_ADMIN', TRUE, NOW(), NOW())
         RETURNING id, email, role, must_change_password`,
        [email, passwordHash]
      );

      user = inserted.rows[0];
    }

    // Revoke any old sessions for this account so the temporary credential
    // starts from a clean session state.
    await db.query(
      `DELETE FROM sessions WHERE user_id = $1`,
      [user.id]
    );

    await db.query("COMMIT");

    return {
      success: true,
      userId: user.id,
      email: user.email,
      role: user.role,
      mustChangePassword: user.must_change_password
    };
  } catch (error) {
    await db.query("ROLLBACK");
    throw error;
  }
}
